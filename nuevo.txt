package com.fredy.crudProductos.controller;

import com.fredy.crudProductos.model.Producto;
import com.fredy.crudProductos.service.ProductoService;
import jakarta.validation.Valid;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.validation.BindingResult;
import org.springframework.web.bind.annotation.*;

import java.util.List;

/**
 * Controlador principal.
 * Recibe las peticiones HTTP del navegador (GET, POST) y decide
 * qué hacer: llamar al service y devolver una vista (HTML) al usuario.
 */
@Controller // Indica que esta clase maneja vistas (HTML), no solo datos JSON
@RequestMapping("/") // Todas las rutas de esta clase empiezan con /productos
public class ProductoController {

    @Autowired
    private ProductoService productoService;

    /**
     * RUTA PRINCIPAL: GET /productos
     * Muestra la página con la tabla de todos los productos y el formulario.
     * También soporta un parámetro opcional "buscar" para filtrar la tabla.
     */
    @GetMapping
    public String index(@RequestParam(value = "buscar", required = false) String buscar,
                         Model model) {

        List<Producto> productos = productoService.buscar(buscar);

        // Agregamos datos al "Model": Thymeleaf los usará para pintar el HTML
        model.addAttribute("productos", productos);
        model.addAttribute("producto", new Producto()); // Objeto vacío para el formulario de registro
        model.addAttribute("buscar", buscar); // Para mantener el texto en el buscador

        return "index"; // Nombre de la plantilla: templates/index.html
    }

    /**
     * RUTA: POST /productos/guardar
     * Recibe los datos del formulario (crear O actualizar) y los guarda.
     * Si el formulario trae un id oculto, se actualiza; si no, se crea nuevo.
     */
    @PostMapping("/guardar")
    public String guardar(@Valid @ModelAttribute("producto") Producto producto,
                           BindingResult resultado,
                           Model model) {

        // Si hay errores de validación (ej: campo vacío), volvemos a mostrar el formulario con errores
        if (resultado.hasErrors()) {
            model.addAttribute("productos", productoService.listarTodos());
            return "index";
        }

        productoService.guardar(producto);
        return "redirect:/productos"; // Redirige a la lista actualizada
    }

    /**
     * RUTA: GET /productos/editar/{id}
     * Carga los datos de un producto existente dentro del formulario para editarlo.
     */
    @GetMapping("/editar/{id}")
    public String editar(@PathVariable Long id, Model model) {
        Producto producto = productoService.buscarPorId(id);

        model.addAttribute("producto", producto);
        model.addAttribute("productos", productoService.listarTodos());

        return "index";
    }

    /**
     * RUTA: GET /productos/eliminar/{id}
     * Elimina un producto por su id y regresa a la lista.
     */
    @GetMapping("/eliminar/{id}")
    public String eliminar(@PathVariable Long id) {
        productoService.eliminar(id);
        return "redirect:/productos";
    }
}

package com.fredy.crudProductos.model;

import jakarta.persistence.*;

/**
 * Clase Producto.
 *
 * Representa la tabla "productos" en la base de datos.
 */
@Entity
@Table(name = "productos")
public class Producto {

    // ID autogenerado por PostgreSQL.
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    // Nombre del producto.
    private String nombre;

    // Código del producto.
    private String codigo;

    // Pasillo donde está ubicado el producto.
    private String pasillo;

    // Precio del producto.
    private double precio;

    // Cantidad disponible del producto.
    private int cantidad;


    // Constructor vacío (Spring Boot lo necesita).
    public Producto() {
    }


    // Constructor con parámetros.
    public Producto(String nombre, String codigo, String pasillo,
                    double precio, int cantidad) {

        this.nombre = nombre;
        this.codigo = codigo;
        this.pasillo = pasillo;
        this.precio = precio;
        this.cantidad = cantidad;
    }


    // ---------------- GETTERS Y SETTERS ----------------


    // ID
    public Long getId() {
        return id;
    }

    public void setId(Long id) {
        this.id = id;
    }


    // Nombre
    public String getNombre() {
        return nombre;
    }

    public void setNombre(String nombre) {
        this.nombre = nombre;
    }


    // Código
    public String getCodigo() {
        return codigo;
    }

    public void setCodigo(String codigo) {
        this.codigo = codigo;
    }


    // Pasillo
    public String getPasillo() {
        return pasillo;
    }

    public void setPasillo(String pasillo) {
        this.pasillo = pasillo;
    }


    // Precio
    public double getPrecio() {
        return precio;
    }

    public void setPrecio(double precio) {
        this.precio = precio;
    }


    // Cantidad
    public int getCantidad() {
        return cantidad;
    }

    public void setCantidad(int cantidad) {
        this.cantidad = cantidad;
    }

}

package com.fredy.crudProductos.repository;

import com.fredy.crudProductos.model.Producto;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.stereotype.Repository;

import java.util.List;

/**
 * Repositorio de Producto.
 * Al extender JpaRepository<Producto, Long> heredamos GRATIS métodos como:
 * save(), findAll(), findById(), deleteById(), etc. sin escribir SQL.
 *
 * <Producto, Long> significa: trabaja con la entidad Producto,
 * y su clave primaria (id) es de tipo Long.
 */
@Repository // Le dice a Spring que esta interfaz es un componente de acceso a datos
public interface ProductoRepository extends JpaRepository<Producto, Long> {

    /**
     * Método de búsqueda personalizado.
     * Spring Data JPA genera automáticamente la consulta SQL
     * solo con el nombre del método, siguiendo su convención de nombres:
     * "buscar productos donde el nombre O el código contengan el texto,
     *  sin importar mayúsculas/minúsculas (IgnoreCase)".
     */
    List<Producto> findByNombreContainingIgnoreCaseOrCodigoContainingIgnoreCase(
            String nombre, String codigo);
}

package com.fredy.crudProductos.service;

import com.fredy.crudProductos.model.Producto;
import com.fredy.crudProductos.repository.ProductoRepository;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;

import java.util.List;

/**
 * Clase de servicio.
 * Aquí va la LÓGICA DE NEGOCIO del CRUD.
 * El controlador NUNCA debe hablar directamente con el repositorio;
 * siempre pasa por el service. Esto mantiene el código ordenado y reutilizable.
 */
@Service // Marca esta clase como un componente de servicio de Spring
public class ProductoService {

    // Inyección de dependencia: Spring nos entrega automáticamente
    // una instancia lista para usar del repositorio.
    @Autowired
    private ProductoRepository productoRepository;

    // ---------- CREATE / UPDATE ----------
    /**
     * Guarda un producto nuevo o actualiza uno existente.
     * Si el producto trae un id, JPA hace UPDATE; si no trae id, hace INSERT.
     */
    public Producto guardar(Producto producto) {
        return productoRepository.save(producto);
    }

    // ---------- READ (todos) ----------
    /**
     * Devuelve la lista completa de productos registrados.
     */
    public List<Producto> listarTodos() {
        return productoRepository.findAll();
    }

    // ---------- READ (uno solo por id) ----------
    /**
     * Busca un producto por su id.
     * Devuelve null si no existe (lo controlamos en el controller).
     */
    public Producto buscarPorId(Long id) {
        return productoRepository.findById(id).orElse(null);
    }

    // ---------- READ (búsqueda por texto) ----------
    /**
     * Busca productos cuyo nombre o código contenga el texto ingresado.
     * Si el texto viene vacío, devuelve todos los productos.
     */
    public List<Producto> buscar(String texto) {
        if (texto == null || texto.isBlank()) {
            return listarTodos();
        }
        return productoRepository
                .findByNombreContainingIgnoreCaseOrCodigoContainingIgnoreCase(texto, texto);
    }

    // ---------- DELETE ----------
    /**
     * Elimina un producto según su id.
     */
    public void eliminar(Long id) {
        productoRepository.deleteById(id);
    }
}

package com.fredy.crudProductos;

/**
 * Clase principal de la aplicación.
 * Es el punto de entrada (el "main") que arranca todo el proyecto Spring Boot.
 *
 * La anotación @SpringBootApplication combina tres anotaciones:
 * - @Configuration: permite definir configuraciones dentro de esta clase.
 * - @EnableAutoConfiguration: Spring configura automáticamente muchas cosas
 *   (servidor web, conexión a BD, etc.) según las dependencias del pom.xml.
 * - @ComponentScan: le dice a Spring que escanee este paquete y subpaquetes
 *   (controller, service, repository, model) para encontrar y registrar
 *   todas las clases anotadas (@Controller, @Service, @Repository, etc.)
 */
@SpringBootApplication
public class CrudProductosApplication {

    // Método main: es el que Java ejecuta primero al iniciar el programa
    public static void main(String[] args) {
        // Este método arranca el servidor embebido (Tomcat) y todo el contexto de Spring
        SpringApplication.run(CrudProductosApplication.class, args);
    }

}
